224    async def _make_request(
225            self,
226            method: str,
227            url: str,
228            json: Any = None,
229            params: Optional[JsonDict] = None,
230    ) -> Any:
231        """Internal method to execute the actual request with retry logic."""
232
233        async def _do_request() -> Any:
234            async with self._session.request(
235                    method,
236                    url,
237                    json=json,
238                    params=params,
239                    raise_for_status=False,
240            ) as response:
241                if response.status >= 400:
242                    await self._handle_error_response(response)
243
244                if response.status == 204:
245                    return True
246
247                try:
248                    await response.json()
249                    return response
250                except aiohttp.ContentTypeError:
251                    return await response.text()
252                except Exception as exception:
253                    raise APIError(
254                        f"Failed to parse response: {exception}", response.status
255                    )
256
257        return await self._retry_request(_do_request, attempts=self._retry_attempts)

259    @staticmethod
260    async def _retry_request(
261            request_func: Callable[[], Awaitable[T]],
262            *,
263            attempts: int = DEFAULT_RETRY_ATTEMPTS,
264            delay: float = DEFAULT_RETRY_DELAY,
265    ) -> T:
266        """
267        Execute request with retry logic.
268
269        Args:
270            request_func: Async function to execute
271            attempts: Maximum number of retry attempts
272            delay: Delay between retries in seconds
273
274        Returns:
275            Response from the successful request
276
277        Raises:
278            APIError: If all retry attempts fail
279        """
280        last_error = None
281
282        for attempt in range(attempts):
283            try:
284                return await request_func()
285            except (aiohttp.ClientError, APIError) as error:
286                last_error = error
287
288                # Don't retry if it's not a retriable error
289                if isinstance(error, APIError) and (
290                        error.status_code not in RETRY_STATUS_CODES
291                ):
292                    raise
293
294                # Don't sleep on the last attempt
295                if attempt < attempts - 1:
296                    await asyncio.sleep(delay * (attempt + 1))
297
298        raise APIError(
299            f"Request failed after {attempts} attempts: {last_error}",
300            getattr(last_error, "status_code", None),
301        )

303    def _build_url(self, endpoint: str) -> str:
304        """Build and validate the full URL for the API request."""
305        if not isinstance(endpoint, str):
306            raise ValueError("Endpoint must be a string")
307
308        url = f"{self._api_url}/{endpoint.lstrip('/')}"
309        parsed_url = urlparse(url)
310
311        if not all([parsed_url.scheme, parsed_url.netloc]):
312            raise ValueError(f"Invalid URL: {url}")
313
314        return url

316    def _get_ssl_context(self) -> Optional[Fingerprint]:
317        """Create an SSL context if a certificate fingerprint is provided."""
318        if not self._cert_sha256:
319            return None
320
321        try:
322            return Fingerprint(binascii.unhexlify(self._cert_sha256))
323        except binascii.Error as validation_error:
324            raise ValueError(
325                f"Invalid certificate SHA256: {self._cert_sha256}"
326            ) from validation_error
327        except Exception as exception:
328            raise OutlineError("Failed to create SSL context") from exception

Decorator for rate-limiting API methods.

Arguments:
  • rate_per_second: Requests per second (defaults to client's rate)
  • burst: Maximum burst size (defaults to client's burst)
