218    async def _make_request(
219        self,
220        method: str,
221        url: str,
222        json: Any = None,
223        params: Optional[JsonDict] = None,
224    ) -> Any:
225        """Internal method to execute the actual request with retry logic."""
226
227        async def _do_request() -> Any:
228            async with self._session.request(
229                method,
230                url,
231                json=json,
232                params=params,
233                raise_for_status=False,
234            ) as response:
235                if response.status >= 400:
236                    await self._handle_error_response(response)
237
238                if response.status == 204:
239                    return True
240
241                try:
242                    return response
243                except Exception as exception:
244                    raise APIError(
245                        f"Failed to process response: {exception}", response.status
246                    )
247
248        return await self._retry_request(_do_request, attempts=self._retry_attempts)

250    @staticmethod
251    async def _retry_request(
252        request_func: Callable[[], Awaitable[T]],
253        *,
254        attempts: int = DEFAULT_RETRY_ATTEMPTS,
255        delay: float = DEFAULT_RETRY_DELAY,
256    ) -> T:
257        """
258        Execute request with retry logic.
259
260        Args:
261            request_func: Async function to execute
262            attempts: Maximum number of retry attempts
263            delay: Delay between retries in seconds
264
265        Returns:
266            Response from the successful request
267
268        Raises:
269            APIError: If all retry attempts fail
270        """
271        last_error = None
272
273        for attempt in range(attempts):
274            try:
275                return await request_func()
276            except (aiohttp.ClientError, APIError) as error:
277                last_error = error
278
279                # Don't retry if it's not a retriable error
280                if isinstance(error, APIError) and (
281                    error.status_code not in RETRY_STATUS_CODES
282                ):
283                    raise
284
285                # Don't sleep on the last attempt
286                if attempt < attempts - 1:
287                    await asyncio.sleep(delay * (attempt + 1))
288
289        raise APIError(
290            f"Request failed after {attempts} attempts: {last_error}",
291            getattr(last_error, "status_code", None),
292        )

294    def _build_url(self, endpoint: str) -> str:
295        """Build and validate the full URL for the API request."""
296        if not isinstance(endpoint, str):
297            raise ValueError("Endpoint must be a string")
298
299        url = f"{self._api_url}/{endpoint.lstrip('/')}"
300        parsed_url = urlparse(url)
301
302        if not all([parsed_url.scheme, parsed_url.netloc]):
303            raise ValueError(f"Invalid URL: {url}")
304
305        return url

307    def _get_ssl_context(self) -> Optional[Fingerprint]:
308        """Create an SSL context if a certificate fingerprint is provided."""
309        if not self._cert_sha256:
310            return None
311
312        try:
313            return Fingerprint(binascii.unhexlify(self._cert_sha256))
314        except binascii.Error as validation_error:
315            raise ValueError(
316                f"Invalid certificate SHA256: {self._cert_sha256}"
317            ) from validation_error
318        except Exception as exception:
319            raise OutlineError("Failed to create SSL context") from exception
